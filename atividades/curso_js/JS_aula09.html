Introdução

O que são "Objetos Literais"?

book_2 Conceito

Objetos literais são estruturas de dados fundamentais em JavaScript que permitem armazenar coleções de pares chave-valor. Eles são ideais para representar entidades do mundo real com múltiplas propriedades.



Sintaxe / Exemplos:

// Objeto vazio
const objetoVazio = {};

// Objeto com propriedades
const pessoa = {
    nome: "Ana Silva",
    idade: 28,
    profissao: "Desenvolvedora",
    ativo: true
};

// Propriedades podem ter diferentes tipos de dados
const produto = {
    id: 101,
    nome: "Notebook",
    preco: 3500.00,
    emEstoque: true,
    especificacoes: {
            marca: "Dell",
            memoria: "16GB"
    }
};


Características Importantes:

  • Chaves (propriedades): podem ser strings ou símbolos
  • Valores: podem ser qualquer tipo de dado (strings, números, booleanos, arrays, outros objetos, funções)
  • Sintaxe: pares chave: valor separados por vírgulas
  • Objetos podem conter métodos (funções como propriedades)
const calculadora = {
    valor: 0,
    somar: function(num) {
                this.valor += num;
                return this.valor;
            },
    // Sintaxe moderna (ES6+)
    subtrair(num) {
                this.valor -= num;
                return this.valor;
            }
};

Acesso e Modificação de Propriedades

book_2 Acessando Propriedades

Existem duas formas principais de acessar propriedades:

const usuario = {
    nome: "Carlos",
    email: "carlos@email.com",
    idade: 35
};

// 1. Notação de ponto (mais comum)
console.log(usuario.nome);  // "Carlos"
console.log(usuario.email); // "carlos@email.com"

// 2. Notação de colchetes (útil para propriedades dinâmicas)
console.log(usuario["idade"]); // 35

// Propriedades dinâmicas
const propriedade = "email";
console.log(usuario[propriedade]); // "carlos@email.com"


book_2 Modificando Propriedades

const carro = {

visibility Exemplo

    // Método para cadastrar novo tutor
    cadastrarTutor(nome, telefone, email) {
        const novoTutor = {
            id: this.tutores.length + 1,
            nome,
            telefone,
            email,
            pets: []
        };
        this.tutores.push(novoTutor);
        return novoTutor;
    },

    // Método para adicionar pet a um tutor
    adicionarPet(tutorId, dadosPet) {
        const tutor = this.tutores.find(t => t.id === tutorId);
        if (!tutor) {
            return { erro: "Tutor não encontrado" };
        }
        const novoPet = {
            id: Date.now(), // ID único baseado em timestamp
            ...dadosPet
        };
        tutor.pets.push(novoPet);
        return novoPet;
    },

    // Método para agendar consulta
    agendarConsulta(tutorId, petId, data, horario, veterinario,
                    motivo, observacoes = "") {
        const tutor = this.tutores.find(t => t.id === tutorId);
        if (!tutor) {
            return { erro: "Tutor não encontrado" };
        }
        const pet = tutor.pets.find(p => p.id === petId);
        if (!pet) {
            return { erro: "Pet não encontrado" };
        }
        const novaConsulta = {
            id: this.consultas.length + 1,
            tutorId,
            petId,
            data,
            horario,
            veterinario,
            motivo,
            status: "agendada",
            observacoes
        };  
        this.consultas.push(novaConsulta);
        return novaConsulta;
    },

    // Método para listar consultas por data
    listarConsultasPorData(data) {
        return this.consultas.filter(c => 
                        c.data === data && c.status === "agendada")
                    .map(consulta => {
                            const tutor = this.tutores.find(t => t.id === consulta.tutorId);
                            const pet = tutor.pets.find(p => p.id === consulta.petId);
                            return {
                                horario: consulta.horario,
                                tutor: tutor.nome,
                                pet: pet.nome,
                                especie: pet.especie,
                                veterinario: consulta.veterinario,
                                motivo: consulta.motivo
                                };
                            })
                        .sort((a, b) => 
                                a.horario.localeCompare(b.horario));
                    },
    
    // Método para listar pets de um tutor
    listarPetsPorTutor(tutorId) {
        const tutor = this.tutores.find(t => t.id === tutorId);
        if (!tutor) {
            return { erro: "Tutor não encontrado" };
        }
        return {
            tutor: tutor.nome,
            pets: tutor.pets
        };
    },

    // Método para cancelar consulta
    cancelarConsulta(consultaId) {
        const consulta = this.consultas.find(c => c.id === consultaId);
        if (!consulta) {
            return { erro: "Consulta não encontrada" };
        }
        consulta.status = "cancelada";
        return { 
            mensagem: "Consulta cancelada com sucesso", consulta 
        };
    },

    // Método para buscar próximas consultas de um pet
    proximasConsultasPet(petId) {
        const hoje = new Date().toISOString().split('T')[0];
        return this.consultas
                    .filter(c => 
                        c.petId === petId && c.data >= hoje && c.status === "agendada")
                    .sort((a, b) => {
                        if (a.data === b.data) {
                            return a.horario.localeCompare(b.horario);
                        }
                        return a.data.localeCompare(b.data);
                    } );
    },

    // Método para gerar relatório do dia
    relatorioDia(data) {
        const consultasDia = this.listarConsultasPorData(data);
        return {
            data,
            totalConsultas: consultasDia.length,
            consultas: consultasDia
        };  
    }
}; // fim da aplicação
