exercise Exercício 1:

Objetivo: Crie um programa que receba uma lista de números inteiros e retorne:

  • Uma tupla contendo todos os números que são divisíveis por 7
  • Uma lista contendo todos os númes ímpares
  • Um dicionário contendo a quantidade de números informada e a soma deles


# Lista de exemplo
lista_numeros = [4, 6, 3, 28, 11, 42, 9, 12, 14,
                 7, 1, 21, 5, 49, 35, 8, 15, 63, 20, 
                 2, 70, 17, 13, 10, 56, 16, 18, 19]

# Lista para armazenar números divisíveis por 7
# (depois vira uma tupla)
divisiveis_por_7 = []

# Lista para armazenar números ímpares
numeros_impares = []

# Processar cada número da lista
for numero in lista_numeros:
    # Verificar se é divisível por 7
    if numero % 7 == 0:
        divisiveis_por_7.append(numero)
    
    # Verificar se é ímpar
    if numero % 2 != 0:
        numeros_impares.append(numero)

# Converter lista de divisíveis por 7 em tupla
tupla_div7 = tuple(divisiveis_por_7)

# Gerar dicionário com contagem e soma de elementos
dicionario_info = {
        "quantidade": len(lista_numeros),
        "soma": sum(lista_numeros)
    }

# Exibir resultados
print("\nANÁLISE DE NÚMEROS")
print("=" * 60)
print(f"Lista original:\n{lista_numeros}")
print("-" * 60)
print(f"TUPLA - Divisíveis por 7: {tupla_div7}")
print(f"LISTA - Números ímpares:  {numeros_impares}")
print(f"DICIONÁRIO - Informações gerais:")
print(f"   Quantidade de números: {dicionario_info['quantidade']}")
print(f"   Soma de todos os números: {dicionario_info['soma']}")
print("=" * 60)

exercise Exercício 2:

Objetivo: Crie uma função que receba uma frase (string) e retorna um dicionário onde as chaves são as palavras da frase e os valores são a quantidade de vezes que cada palavra aparece.


DICA 1: Use um set para verificar se a palavra já foi contada (lembre-se que sets não permitem elementos repetidos)


DICA 2: Para transformar uma frase em uma lista de paavras você pode usar o comando: lista_palavras = frase.lower().split()



def contar_palavras(frase):
    palavras = frase.lower().split()
    palavras_contadas = set()
    contagem = {}

    for palavra in palavras:
        if palavra in palavras_contadas:
            # Palavra já existe na contagem, incrementar contador
            contagem[palavra] += 1
        else:
            # Primeira vez que vemos esta palavra
            palavras_contadas.add(palavra)
            contagem[palavra] = 1
    
    return contagem

# Testando a função
frase1 = "python é incrível python é poderoso e python é divertido"
print(contar_palavras(frase1))

frase2 = "o rato roeu a roupa do rei de roma o rato roeu"
print(contar_palavras(frase2))

frase3 = "a programação em python é legal python facilita a programação"
print(contar_palavras(frase3))

exercise Exercício 3:

Objetivo: Crie um sistema que gerencie notas de alunos. O programa deve:

  • Usar um dicionário onde as chaves são nomes dos alunos e os valores são listas de notas
  • Calcular a média de cada aluno
  • Retornar uma tupla com (nome, média, situação) para cada aluno. A situação deve ser: "Aprovado" se média ≥ 7, "Recuperação" se 5 ≥ média < 7, "Reprovado" se média < 5

Exemplo de entrada:

alunos = {
    "Ana": [8.5, 9.0, 7.5],
    "Bruno": [6.0, 5.5, 6.5],
    "Carlos": [4.0, 3.5, 4.5]
}
                        


alunos_turma = {
    "Ana Silva": [8.5, 9.0, 7.5, 8.0],
    "Bruno Costa": [6.0, 5.5, 6.5, 6.0],
    "Carlos Mendes": [4.0, 3.5, 4.5, 4.0],
    "Diana Santos": [9.0, 9.5, 10.0, 9.0],
    "Eduardo Lima": [7.0, 7.5, 7.0, 7.5],
    "Fernanda Souza": [5.0, 5.5, 5.0, 6.0],
    "Gabriel Rocha": [3.0, 4.0, 3.5, 3.0],
    "Helena Alves": [8.0, 8.5, 9.0, 8.5]
}

# Processar cada aluno
for nome, notas in alunos_turma.items():
    # Calcular média
    media = sum(notas) / len(notas)
    
    # Determinar situação
    if media >= 7:
        situacao = "Aprovado"
    elif media >= 5:
        situacao = "Recuperação"
    else:
        situacao = "Reprovado"
    
    # Criar tupla e exibir
    tupla_aluno = (nome, media, situacao)
    print(tupla_aluno)

exercise Exercício 6: DESAFIO

Você tem transações bancárias armazenadas em uma lista, onde cada elemento da lista é uma transação bancária em forma de dicionário. Sua aplicação precisa criar um pipeline de processamento usando map(), filter() e reduce() atendendo as seguintes demandas:

  1. Use filter() para separar depósitos (valores positivos) e retiradas (valores negativos)
  2. Use map() para aplicar taxa de 2% sobre retiradas
  3. Use reduce() para calcular o saldo final
  4. Use reduce() para encontrar a maior transação (em valor absoluto)
  5. Use filter() + reduce() para contar transações acima de R$ 1000

Exemplo de entrada:

transacoes = [
    {"id": 1, "tipo": "depósito", "valor": 1500.00, "data": "2024-01-15"},
    {"id": 2, "tipo": "retirada", "valor": -200.00, "data": "2024-01-16"},
    {"id": 3, "tipo": "depósito", "valor": 3000.00, "data": "2024-01-17"},
    {"id": 4, "tipo": "retirada", "valor": -1500.00, "data": "2024-01-18"},
    {"id": 5, "tipo": "retirada", "valor": -100.00, "data": "2024-01-19"},
    {"id": 6, "tipo": "depósito", "valor": 500.00, "data": "2024-01-20"}
]
                        


from functools import reduce

# 1. Dados de exemplo
transacoes = [
    {"id": 1, "tipo": "depósito", "valor": 1500.00, "data": "2024-01-15"},
    {"id": 2, "tipo": "retirada", "valor": -200.00, "data": "2024-01-16"},
    {"id": 3, "tipo": "depósito", "valor": 3000.00, "data": "2024-01-17"},
    {"id": 4, "tipo": "retirada", "valor": -1500.00, "data": "2024-01-18"},
    {"id": 5, "tipo": "retirada", "valor": -100.00, "data": "2024-01-19"},
    {"id": 6, "tipo": "depósito", "valor": 500.00, "data": "2024-01-20"}
]

# --- 1. Separação de Depósitos e Retiradas (filter()) ---
# Critério: tipo de transação

depositos = list(filter(lambda tran: tran['tipo'] == "depósito", transacoes))
retiradas = list(filter(lambda tran: tran['tipo'] == 'retirada', transacoes))

print("--- 1. SEPARAÇÃO (FILTER) ---")
print(f"Total de Depósitos: {len(depositos)}")
print(f"Total de Retiradas: {len(retiradas)}")
print("-" * 30)


# --- 2. Aplicação de Taxa sobre Retiradas (map()) ---
def aplicar_taxa(tran):
    if tran['tipo'] == 'retirda':
        valor_ajustado = ret['valor'] * 1.02
    else:
        valor_ajustado = ret['valor']
    return {'id': ret['id'], 
            'tipo: ret['tipo'], 
            'valor': round(valor_ajustado,2),
            'data': ret['data']}

transacoes_com_taxa = list(map(aplicar_taxa, transacoes))

print("--- 2. APLICAÇÃO DE TAXA (MAP) ---")
for trans in transacoes_com_taxa:
    print(f"ID: {trans['id']:2} | Tipo: {trans['tipo']:9} | Valor com taxa: R$ {trans['valor']:.2f}")
print("-" * 30)

# --- 3. Cálculo do Saldo Final (reduce()) ---

saldo_final = reduce(lambda acumulador, transacao: acumulador + transacao['valor'],
                     transacoes_com_taxa,
                     0.0) # O valor inicial do acumulador é 0.0

print("--- 3. SALDO FINAL (REDUCE) ---")
print(f"Saldo Final Calculado: R$ {saldo_final:.2f}")
print("-" * 30)

# --- 4. Maior Transação em Valor Absoluto (reduce()) ---

def maior_transacao(t1, t2):
    """Compara duas transações e retorna a de maior valor absoluto."""
    if abs(t1['valor']) > abs(t2['valor']):
        return t1
    else: 
        return t2

# A função reduce retorna o dicionário completo da maior transação
maior_transacao_dict = reduce(maior_transacao, transacoes)

print("--- 4. MAIOR TRANSAÇÃO (REDUCE) ---")
print(f"ID da Maior Transação (Absoluta): {maior_transacao_dict['id']}")
print(f"Valor: R$ {maior_transacao_dict['valor']:.2f}")
print("-" * 30)


# --- 5. Contagem de Transações Acima de R$ 1000 (filter() + reduce()) ---
# filtar as transações desejadas
transacoes_acima_mil = list(filter(lambda t: abs(t['valor']) > 1000, transacoes))

# Contar o número de transações (reduce())
contagem_transacoes = reduce(lambda count, tran: count + 1,
                             transacoes_acima_mil,
                             0) # O valor inicial do acumulador é 0

print("--- 5. CONTAGEM ACIMA DE R$ 1000 (FILTER + REDUCE) ---")
print(f"Transações acima de R$ 1000 (abs): {contagem_transacoes}")
print("-" * 30)
